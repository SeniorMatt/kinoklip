# excludelist.txt -- which shared libraries the Linux bundle must NOT carry.
#
# Read by scripts/package_linux.sh (dependency walker) and scripts/check_bundle.sh.
#
# Format: one entry per line, "<kind> <soname-glob>", '#' starts a comment.
#   host      never bundled; the library must come from the user's system. The walker
#             does not descend into it either, so its own dependencies are not bundled. Every
#             entry states why on its own line ("host <soname> # reason"): scripts/check_bundle.sh
#             fails without one, and also if a bundled file needs anything host-only but glibc.
#   fallback  prefer the host copy, but ship one in usr/lib/fallback/. The launcher links
#             only the ones the host lacks into usr/lib/fallback-active/ (searched via
#             RUNPATH), so a system copy always wins when it exists.
#   newer     like fallback, but the bundled copy is also used when the host has an OLDER
#             one: compared by the highest GLIBCXX_/GCC_ version tag inside the file, or else
#             by the version in the real file name (libfontconfig.so.1.17.0). Only for
#             libraries that are backwards compatible, so the newest copy is always safe --
#             also for host libraries loaded into the same process.
# A current Arch build references symbols that older distributions' copies lack (found with
# LD_BIND_NOW=1 LD_WARN=yes on Linux Mint 22): fontconfig 2.17 (FcConfigSetDefaultSubstitute,
# used by pango/libass) and wayland-client 1.23+ (wl_display_create_queue_with_name, used
# by KF6GuiAddons). Both keep their ABI stable, so they are "newer" fallbacks.
#
# Rule of thumb: a library is "fallback" unless it truly cannot work from the bundle. A distro
# can lack any library or name it differently (GitHub issue 1: Fedora's libbz2.so.1 instead of
# libbz2.so.1.0; a Wayland-only GNOME install has no libX11 guarantee; a distro without systemd
# has no libsystemd), and "every system has this one" has been wrong before. A fallback costs
# nothing on a system that has the library, since the host copy wins: it is loaded only where the
# app could not have started otherwise, and there no host library can want another copy of it.
# Why the host copy wins (a bundled copy on every system would be wrong): the host's graphics
# and font stack (Mesa, LLVM, freetype, ...) links these too, and a bundled older copy would be
# loaded first and shadow the newer one that host library was built against ("undefined symbol").
# "host" is only for what cannot work from the bundle at all: glibc, which has its own mechanism
# (below), and GPU drivers, which belong to the installed kernel module and hardware.
#
# Build host: the Arch Linux container kk-arch (glibc 2.44, GCC 16 at the time of writing).
# Target: KDE Linux (Arch based, Wayland); other current distributions should work as long
# as their glibc is at least the one the bundle needs (scripts/check_bundle.sh prints it).

# ---- C runtime (glibc): taken from the host. Never walked into the bundle; package_linux.sh
# ships the build host's glibc separately in usr/lib/glibc/, which AppRun only uses (via its
# own ld.so) when the host glibc is older than the bundle needs ----
host ld-linux-x86-64.so.2 # glibc: the loader itself, see usr/lib/glibc
host linux-vdso.so.1 # glibc: provided by the kernel, not a file
host libc.so.6 # glibc: one per process, see usr/lib/glibc
host libm.so.6 # glibc: one per process, see usr/lib/glibc
host libpthread.so.0 # glibc: one per process, see usr/lib/glibc
host libdl.so.2 # glibc: one per process, see usr/lib/glibc
host librt.so.1 # glibc: one per process, see usr/lib/glibc
host libresolv.so.2 # glibc: one per process, see usr/lib/glibc
host libutil.so.1 # glibc: one per process, see usr/lib/glibc
host libanl.so.1 # glibc: one per process, see usr/lib/glibc
host libnsl.so.1 # glibc: one per process; nothing bundled needs it
host libmvec.so.1 # glibc: one per process, see usr/lib/glibc
host libBrokenLocale.so.1 # glibc: one per process; nothing bundled needs it

# ---- C++/GCC runtime: backwards compatible, so the NEWEST copy must be loaded. An Arch build
# needs a newer libstdc++ than e.g. Ubuntu 24.04 has (GLIBCXX_3.4.36 vs 3.4.33): ship ours
# as "newer" fallback; on a host with the same or a newer GCC runtime (KDE Linux) the host
# copy wins, so host libraries (Mesa/LLVM) never see an older libstdc++ than they need.
newer libstdc++.so.6
newer libgcc_s.so.1
fallback libatomic.so.1
# OpenMP runtime (x265/libvpx); part of gcc-libs on Arch, but tiny, so keep a fallback.
fallback libgomp.so.1

# ---- GPU drivers: must match the installed kernel module and hardware; the loaders below find
# them on the host at run time. ----
host libGLX_*.so.0 # GPU driver: vendor library (Mesa, NVIDIA) chosen by the glvnd dispatcher
host libEGL_*.so.0 # GPU driver: vendor library (Mesa, NVIDIA) chosen by the glvnd dispatcher
host libgbm.so.1 # GPU driver: loads the host's Mesa/NVIDIA buffer backends, ABI tied to them
host libdrm_*.so.* # GPU driver: per-chip helpers of the host's Mesa (amdgpu, intel, nouveau, ...)
host libcuda.so.1 # GPU driver: part of the NVIDIA driver, tied to its kernel module
host libnvidia-*.so* # GPU driver: part of the NVIDIA driver, tied to its kernel module
host libnvcuvid.so.1 # GPU driver: part of the NVIDIA driver, tied to its kernel module

# ---- GPU front ends, no driver inside: the glvnd dispatchers and the Vulkan loader look for
# the host's drivers themselves (/usr/share/glvnd/egl_vendor.d, /usr/share/vulkan/icd.d, the
# cache for libGLX_*). A host lacking one has no working driver behind it, but must still start,
# drawing in software. Fedora ships libOpenGL in a package GNOME does not pull in, for example.
fallback libGL.so.1
fallback libEGL.so.1
fallback libGLX.so.0
fallback libOpenGL.so.0
fallback libGLdispatch.so.0
fallback libGLESv2.so.2
fallback libvulkan.so.1
fallback libdrm.so.2
# libva is a "newer" fallback: an Arch FFmpeg calls vaMapBuffer2 (libva 2.21+), which e.g.
# Ubuntu 24.04's libva 2.20 lacks. A newer libva still loads older drivers; when the bundled
# one is used, AppRun points LIBVA_DRIVERS_PATH at the host's driver directory (the bundled
# libva would otherwise look in Arch's /usr/lib/dri only).
# VAAPI/VDPAU front ends: the host libva loads the host iHD/radeonsi driver
# (LIBVA_DRIVERS_PATH is never overridden). FFmpeg links them hard, so if a minimal host
# lacks one (libvdpau is optional on recent Arch since Mesa dropped VDPAU) the bundled copy
# is used instead of failing to start.
newer libva.so.2
newer libva-drm.so.2
newer libva-x11.so.2
newer libva-wayland.so.2
fallback libvdpau.so.1
# OpenCL ICD loader: reads /etc/OpenCL/vendors of the host; ship it in case it is missing.
fallback libOpenCL.so.1
